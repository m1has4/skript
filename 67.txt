local function FB_GET(path)
    local ok, res = pcall(function()
        return request({ Url = FIREBASE_URL .. path .. ".json", Method = "GET" })
    end)
    if not ok or not res or not res.Body then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
    return ok2 and data or nil
end

local function FB_PUT(path, data)
    local ok, res = pcall(function()
        return request({
            Url = FIREBASE_URL .. path .. ".json",
            Method = "PUT",
            Body = HttpService:JSONEncode(data),
            Headers = { ["Content-Type"] = "application/json" }
        })
    end)
    if ok and res then
        if res.StatusCode == 200 then return true end
        if res.StatusCode == 401 or res.StatusCode == 403 then return false, "403" end
    end
    return false, "net"
end

local function FB_DELETE(path)
    pcall(function()
        request({ Url = FIREBASE_URL .. path .. ".json", Method = "DELETE" })
    end)
end

local function GetTargetHead(m)
    local armor = m:FindFirstChild("CharacterArmor")
    if armor then local h = armor:FindFirstChild("Head"); if h and h:IsA("BasePart") then return h end end
    local mh = m:FindFirstChild("Head"); if mh and mh:IsA("BasePart") then return mh end
    local ut = m:FindFirstChild("UpperTorso")
    if ut then local h = ut:FindFirstChild("Head"); if h and h:IsA("BasePart") then return h end end
    return nil
end

local function FindTarget()
    local folder = workspace:FindFirstChild("Characters")
    if not folder then return nil end
    local vs = Camera.ViewportSize
    local center = Vector2.new(vs.X / 2, vs.Y / 2)
    local closest, shortest = nil, AimbotConfig.FOV
    for _, m in pairs(folder:GetChildren()) do
        if m.Name == LocalPlayer.Name then continue end
        if IsHidden(m.Name, "aim") then continue end
        if IsHidden(m.Name, "invis") then continue end
        local head = GetTargetHead(m)
        if head then
            local d3 = (head.Position - Camera.CFrame.Position).Magnitude
            if d3 <= AimbotConfig.MaxDistance then
                local sp, on = Camera:WorldToViewportPoint(head.Position)
                if on then
                    local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d2 <= shortest then shortest = d2; closest = head end
                end
            end
        end
    end
    return closest
end

local HoldActive = false
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if AimbotConfig.HoldMode and input.UserInputType == AimbotConfig.HoldKey then HoldActive = true end
end)
UserInputService.InputEnded:Connect(function(input)
    if AimbotConfig.HoldMode and input.UserInputType == AimbotConfig.HoldKey then HoldActive = false end
end)

local function MoveMouseToTarget(t)
    local sp, on = Camera:WorldToViewportPoint(t.Position)
    if not on then return end
    local mp = UserInputService:GetMouseLocation()
    local dx, dy = sp.X - mp.X, sp.Y - mp.Y

    if AimbotConfig.AimMode == "anti" then
        -- Анти-аим: отводим прицел в СТОРОНУ (перпендикулярно) с большим смещением
        local angle = math.atan2(dy, dx) + math.rad(90)
        local dist = math.sqrt(dx*dx + dy*dy)
        dx = math.cos(angle) * dist
        dy = math.sin(angle) * dist
    end

    if AimbotConfig.Smoothness > 0 then
        dx = dx * (1 - AimbotConfig.Smoothness)
        dy = dy * (1 - AimbotConfig.Smoothness)
    end
    mousemoverel(dx, dy)
end

local function AimbotLoop()
    if not AimbotEnabled then return end
    if AimbotConfig.HoldMode and not HoldActive then return end
    local t = FindTarget()
    if not t then return end

    if AimbotConfig.MoveMouse then
        MoveMouseToTarget(t)
    else
        local cp = Camera.CFrame.Position
        local targetPos = t.Position
        if AimbotConfig.AimMode == "anti" then
            -- Отводим камеру
            local dir = (targetPos - cp).Unit
            local perp = Vector3.new(-dir.Z, 0, dir.X).Unit * 50
            targetPos = targetPos + perp
        end
        local tc = CFrame.lookAt(cp, targetPos)
        if AimbotConfig.Smoothness > 0 then Camera.CFrame = Camera.CFrame:Lerp(tc, 1 - AimbotConfig.Smoothness)
        else Camera.CFrame = tc end
    end
end

    if not folder then return end
    local vs = Camera.ViewportSize
    local center = Vector2.new(vs.X / 2, vs.Y / 2)

    for _, m in pairs(folder:GetChildren()) do
        if m.Name == LocalPlayer.Name then continue end
        -- ⛔ если цель защищена анти-скейлом — пропускаем
        if IsHidden(m.Name, "scale") then continue end
        local head = GetTargetHead(m)
        if head then
            local d3 = (head.Position - Camera.CFrame.Position).Magnitude
            if d3 <= HeadScaleConfig.MaxDistance then
                local sp, on = Camera:WorldToViewportPoint(head.Position)
                if on then
                    local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d2 <= HeadScaleConfig.FOV then
                        -- Сохраняем оригинал один раз
                        if not HeadScaleOriginals[head] then
                            HeadScaleOriginals[head] = head.Size
                        end
                        local orig = HeadScaleOriginals[head]
                        head.Size = orig * HeadScaleConfig.Scale
                    end
                end
            end
        end
    end
end

local function ToggleDeleteMap(state)
    DeleteMapEnabled = state
    if state then
        local mo = workspace:FindFirstChild("map")
        if mo then
            SavedMapObject = mo; SavedMapParent = mo.Parent; SavedMapCFrame = mo:GetPivot(); mo.Parent = nil
        else
            for _, obj in pairs(workspace:GetDescendants()) do
                if obj.Name == "map" or obj.Name == "Map" or obj.Name == "MAP" then
                    SavedMapObject = obj; SavedMapParent = obj.Parent; SavedMapCFrame = obj:GetPivot(); obj.Parent = nil; break
                end
            end
        end
    else
        if SavedMapObject and SavedMapParent then
            SavedMapObject.Parent = SavedMapParent
            SavedMapObject:PivotTo(SavedMapCFrame)
            SavedMapObject = nil; SavedMapParent = nil; SavedMapCFrame = nil
        end
    end
    if UIObjects["Delete Map"] then UIObjects["Delete Map"].SetState(state) end
end

local function RefreshOnline()
    local data = FB_GET("/online")
    if not data then return end
    local now = os.time()
    local list = {}
    for _, info in pairs(data) do
        -- скрытые админы не показываются
        if type(info) == "table" and info.lastSeen and (now - info.lastSeen) < 90 and info.hidden ~= true then
            table.insert(list, info)
        end
    end
    local ord = { admin = 1, tester = 2, user = 3 }
    table.sort(list, function(a, b)
        local oa = ord[a.role or "user"] or 3
        local ob = ord[b.role or "user"] or 3
        if oa ~= ob then return oa < ob end
        return (a.name or "") < (b.name or "")
    end)
    onlineLabel.Text = "Онлайн: " .. #list
    for _, ch in pairs(onlineListFrame:GetChildren()) do if ch:IsA("TextLabel") then ch:Destroy() end end
    for i, info in ipairs(list) do
        local r = Instance.new("TextLabel")
        r.Size = UDim2.new(1, -6, 0, 18)
        r.BackgroundTransparency = 1
        r.TextColor3 = RoleColors[info.role or "user"] or Theme.Text
        r.Font = Enum.Font.GothamMedium
        r.TextSize = 11
        r.TextXAlignment = Enum.TextXAlignment.Left
        r.LayoutOrder = i
        local suffix = ""
        if info.role == "admin" then suffix = "  [ADMIN]"
        elseif info.role == "tester" then suffix = "  [TESTER]" end
        r.Text = "• " .. (info.name or "?") .. suffix
        r.Parent = onlineListFrame
    end
    onlineListFrame.CanvasSize = UDim2.new(0, 0, 0, #list * 21)
end

local AdminRoomBuilt = false

local function BuildAdminRoom()
    if AdminRoomBuilt then return end
    AdminRoomBuilt = true
    AdminRoomElements = {}

    table.insert(AdminRoomElements, CreateSettingTitle("🛡 ADMIN ROOM", RightColumn))

    local function makeAdminToggle(label, field, default)
        local t
        t = CreateToggle(label, RightColumn, function(state)
            local ok, err = FB_PUT("/adminSettings/" .. LocalPlayer.Name .. "/" .. field, state)
            if ok then
                MySettings[field] = state
                AdminSettings[LocalPlayer.Name] = AdminSettings[LocalPlayer.Name] or {}
                AdminSettings[LocalPlayer.Name][field] = state
                -- если включили/выключили скрытие от онлайна — сразу обновим online
                if field == "hideFromOnline" then
                    FB_PUT("/online/" .. myKey .. "/hidden", state)
                end
                print("[Admin] " .. field .. " = " .. tostring(state))
            else
                warn("[Admin] " .. tostring(err))
                t.SetState(not state)
            end
        end, (MySettings[field] == true) or default)
        table.insert(AdminRoomElements, t.Root)
        return t
    end

    makeAdminToggle("Скрыть меня от ESP", "hideFromESP", false)
    makeAdminToggle("Скрыть меня от Aimbot", "hideFromAimbot", false)
    makeAdminToggle("Скрыть меня от Онлайна", "hideFromOnline", false)
    makeAdminToggle("Невидимость (полная)", "invisible", false)
    makeAdminToggle("Анти-аим (отвод)", "antiAim", false)
    makeAdminToggle("Анти-увеличение головы", "antiHeadScale", false)

    -- Функция: сделать себя неуязвимым ко всему сразу
    local masterToggle
    masterToggle = CreateToggle("🔒 Полная защита (всё)", RightColumn, function(state)
        local fields = {"hideFromESP", "hideFromAimbot", "hideFromOnline", "invisible", "antiAim", "antiHeadScale"}
        for _, f in ipairs(fields) do
            local ok = FB_PUT("/adminSettings/" .. LocalPlayer.Name .. "/" .. f, state)
            if ok then
                MySettings[f] = state
                AdminSettings[LocalPlayer.Name] = AdminSettings[LocalPlayer.Name] or {}
                AdminSettings[LocalPlayer.Name][f] = state
            end
        end
        FB_PUT("/online/" .. myKey .. "/hidden", state)
        print("[Admin] Полная защита: " .. tostring(state))
    end, false)
    table.insert(AdminRoomElements, masterToggle.Root)
end

    AdminRoomBuilt = false
    for _, el in pairs(AdminRoomElements) do
        if el and el.Destroy then pcall(function() el:Destroy() end) end
    end
    AdminRoomElements = {}
end
