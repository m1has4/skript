local ETags = {}

local function FB_GET(path)
    local ok, res = pcall(function()
        return request({ Url = FIREBASE_URL .. path .. ".json", Method = "GET" })
    end)
    if not ok or not res or not res.Body then return nil end
    local ok2, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
    return ok2 and data or nil
end

local function FB_GET_Cached(path)
    local headers = {}
    if ETags[path] then headers["If-None-Match"] = ETags[path] end
    local ok, res = pcall(function()
        return request({ Url = FIREBASE_URL .. path .. ".json", Method = "GET", Headers = headers })
    end)
    if not ok or not res then return nil, false end
    if res.StatusCode == 304 then return nil, false end
    if res.Headers then
        ETags[path] = res.Headers["etag"] or res.Headers["ETag"] or ETags[path]
    end
    local ok2, data = pcall(function() return HttpService:JSONDecode(res.Body) end)
    return ok2 and data or nil, true
end

local function FB_PUT(path, data)
    local ok, res = pcall(function()
        return request({
            Url = FIREBASE_URL .. path .. ".json",
            Method = "PUT",
            Body = HttpService:JSONEncode(data),
            Headers = { ["Content-Type"] = "application/json" }
        })
    end)
    if ok and res and res.StatusCode == 200 then
        ETags[path] = nil
        return true
    end
    return false
end

local function FB_DELETE(path)
    pcall(function()
        request({ Url = FIREBASE_URL .. path .. ".json", Method = "DELETE" })
    end)
    ETags[path] = nil
end

local function GetTargetHead(m)
    local armor = m:FindFirstChild("CharacterArmor")
    if armor then local h = armor:FindFirstChild("Head"); if h and h:IsA("BasePart") then return h end end
    local mh = m:FindFirstChild("Head"); if mh and mh:IsA("BasePart") then return mh end
    local ut = m:FindFirstChild("UpperTorso")
    if ut then local h = ut:FindFirstChild("Head"); if h and h:IsA("BasePart") then return h end end
    return nil
end

local function FindTarget()
    local folder = workspace:FindFirstChild("Characters")
    if not folder then return nil end
    local vs = Camera.ViewportSize
    local center = Vector2.new(vs.X / 2, vs.Y / 2)
    local closest, shortest = nil, AimbotConfig.FOV
    for _, m in pairs(folder:GetChildren()) do
        if m.Name == LocalPlayer.Name then continue end
        if IsHidden(m.Name, "aim") or IsHidden(m.Name, "invis") then continue end
        local head = GetTargetHead(m)
        if head then
            local d3 = (head.Position - Camera.CFrame.Position).Magnitude
            if d3 <= AimbotConfig.MaxDistance then
                local sp, on = Camera:WorldToViewportPoint(head.Position)
                if on then
                    local d2 = (Vector2.new(sp.X, sp.Y) - center).Magnitude
                    if d2 <= shortest then shortest = d2; closest = head end
                end
            end
        end
    end
    return closest
end

local HoldActive = false
UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if AimbotConfig.HoldMode and input.UserInputType == AimbotConfig.HoldKey then HoldActive = true end
end)
UserInputService.InputEnded:Connect(function(input)
    if AimbotConfig.HoldMode and input.UserInputType == AimbotConfig.HoldKey then HoldActive = false end
end)

local function AimbotLoop()
    if not AimbotEnabled then return end
    if AimbotConfig.HoldMode and not HoldActive then return end
    local t = FindTarget()
    if not t then return end
    if AimbotConfig.MoveMouse then
        local sp, on = Camera:WorldToViewportPoint(t.Position)
        if not on then return end
        local mp = UserInputService:GetMouseLocation()
        local dx, dy = sp.X - mp.X, sp.Y - mp.Y
        if AimbotConfig.Smoothness > 0 then
            dx = dx * (1 - AimbotConfig.Smoothness)
            dy = dy * (1 - AimbotConfig.Smoothness)
        end
        mousemoverel(dx, dy)
    else
        local cp = Camera.CFrame.Position
        local tc = CFrame.lookAt(cp, t.Position)
        if AimbotConfig.Smoothness > 0 then Camera.CFrame = Camera.CFrame:Lerp(tc, 1 - AimbotConfig.Smoothness)
        else Camera.CFrame = tc end
    end
end

local function StartAimbot()
    RunService:UnbindFromRenderStep("SharedAimbotCamera")
    RunService:BindToRenderStep("SharedAimbotCamera", Enum.RenderPriority.Camera.Value + 1, AimbotLoop)
    UpdateFOVVisuals()
end
local function StopAimbot()
    RunService:UnbindFromRenderStep("SharedAimbotCamera")
    UpdateFOVVisuals()
end

local function IsVisible(targetPart)
    if not SharedESP.UseVisibilityCheck then return true end
    local origin = Camera.CFrame.Position
    local dir = targetPart.Position - origin
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    local filter = { Camera }
    if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
    params.FilterDescendantsInstances = filter
    params.IgnoreWater = true
    local result = workspace:Raycast(origin, dir, params)
    if not result then return true end
    return result.Instance:IsDescendantOf(targetPart.Parent) or result.Instance == targetPart
end

local function CollectVisibleEnemies()
    local folder = workspace:FindFirstChild("Characters")
    if not folder then return {} end
    local result = {}
    for _, m in pairs(folder:GetChildren()) do
        if m.Name == LocalPlayer.Name then continue end
        if IsHidden(m.Name, "invis") then continue end
        local head = GetTargetHead(m)
        if head then
            local d = (head.Position - Camera.CFrame.Position).Magnitude
            if d <= SharedESP.MaxDistance and IsVisible(head) then
                local p = head.Position
                result[m.Name] = {
                    x = math.floor(p.X * 10) / 10,
                    y = math.floor(p.Y * 10) / 10,
                    z = math.floor(p.Z * 10) / 10,
                    t = os.time()
                }
            end
        end
    end
    return result
end

local function SetupBox(box)
    if not box then return end
    box.outline.Thickness = SharedESP.BoxThickness + 1
    box.outline.Color = Color3.new(0, 0, 0)
    box.outline.Filled = false
    box.outline.Visible = false
    box.rect.Thickness = SharedESP.BoxThickness
    box.rect.Color = SharedESP.BoxColor
    box.rect.Filled = false
    box.rect.Visible = false
    box.name.Size = 14
    box.name.Center = true
    box.name.Outline = true

    box.name.Visible = false
end

local function HideBox(box)
    if not box then return end
    box.rect.Visible = false
    box.outline.Visible = false
    box.name.Visible = false
end

local function DrawBoxFor(nick, pos3d)
    local box = boxes[nick]
    if not box then
        box = CreateBox()
        if not box then return end
        SetupBox(box)
        boxes[nick] = box
    end

    local sp, onScreen = Camera:WorldToViewportPoint(pos3d)
    if not onScreen then HideBox(box) return end

    local dist = (pos3d - Camera.CFrame.Position).Magnitude
    local h = math.clamp(2000 / dist, 20, 200)
    local w = h * 0.6

    box.rect.Size = Vector2.new(w, h)
    box.rect.Position = Vector2.new(sp.X - w/2, sp.Y - h/2)
    box.outline.Size = Vector2.new(w + 2, h + 2)
    box.outline.Position = Vector2.new(sp.X - w/2 - 1, sp.Y - h/2 - 1)
    box.name.Position = Vector2.new(sp.X, sp.Y - h/2 - 18)
    box.name.Text = nick

    box.rect.Visible = true
    box.outline.Visible = true
    box.name.Visible = SharedESP.ShowName
end

local function RenderSharedESP()
    if not SharedESP.Enabled then
        for _, box in pairs(boxes) do HideBox(box) end
        return
    end

    local now = os.time()
    local toRender = {}

    for senderId, targets in pairs(sharedData) do
        if senderId == MY_ID or senderId == "_init" then continue end
        if type(targets) ~= "table" then continue end
        for nick, info in pairs(targets) do
            if type(info) ~= "table" then continue end
            if nick == LocalPlayer.Name then continue end
            if IsHidden(nick, "invis") or IsHidden(nick, "esp") then continue end
            local age = now - (info.t or 0)
            if age <= SharedESP.TTL then
                toRender[nick] = Vector3.new(info.x, info.y, info.z)
            end
        end
    end

    for nick, pos in pairs(toRender) do DrawBoxFor(nick, pos) end
    for nick, box in pairs(boxes) do
        if not toRender[nick] then HideBox(box) end
    end
end

local function RefreshOnline()
    local data = FB_GET("/online")
    if not data then return end
    local now = os.time()
    local list = {}
    local stale = {}
    for uid, info in pairs(data) do
        if type(info) == "table" then
            if info.lastSeen and (now - info.lastSeen) > 90 then
                table.insert(stale, uid)
            elseif info.hidden ~= true then
                table.insert(list, info)
            end
        end
    end
    for _, uid in ipairs(stale) do FB_DELETE("/online/" .. uid) end
    local ord = { admin = 1, tester = 2, user = 3 }
    table.sort(list, function(a, b)
        local oa = ord[a.role or "user"] or 3
        local ob = ord[b.role or "user"] or 3
        if oa ~= ob then return oa < ob end
        return (a.name or "") < (b.name or "")
    end)
    onlineLabel.Text = "Онлайн: " .. #list
    for _, ch in pairs(onlineListFrame:GetChildren()) do if ch:IsA("TextLabel") then ch:Destroy() end end
    for i, info in ipairs(list) do
        local r = Instance.new("TextLabel")
        r.Size = UDim2.new(1, -6, 0, 18)
        r.BackgroundTransparency = 1
        r.TextColor3 = RoleColors[info.role or "user"] or Theme.Text
        r.Font = Enum.Font.GothamMedium
        r.TextSize = 11
        r.TextXAlignment = Enum.TextXAlignment.Left
        r.LayoutOrder = i
        local suffix = ""
        if info.role == "admin" then suffix = "  [ADMIN]"
        elseif info.role == "tester" then suffix = "  [TESTER]" end
        r.Text = "• " .. (info.name or "?") .. suffix
        r.Parent = onlineListFrame
    end
    onlineListFrame.CanvasSize = UDim2.new(0, 0, 0, #list * 21)
end

RefreshOnline()

local AdminRoomBuilt = false
local AdminRoomElements = {}

local function BuildAdminRoom()
    if AdminRoomBuilt then return end
    AdminRoomBuilt = true
    AdminRoomElements = {}
    table.insert(AdminRoomElements, CreateSettingTitle("ADMIN ROOM", RightColumn))

    local function makeAdminToggle(label, field, default)
        local t
        t = CreateToggle(label, RightColumn, function(state)
            local ok = FB_PUT("/adminSettings/" .. LocalPlayer.Name .. "/" .. field, state)
            if ok then
                MySettings[field] = state
                AdminSettings[LocalPlayer.Name] = AdminSettings[LocalPlayer.Name] or {}
                AdminSettings[LocalPlayer.Name][field] = state
                if field == "hideFromOnline" then
                    FB_PUT("/online/" .. myKey .. "/hidden", state)
                end
            else
                t.SetState(not state)
            end
        end, (MySettings[field] == true) or default)
        table.insert(AdminRoomElements, t.Root)
        return t
    end

    makeAdminToggle("Скрыть от Shared ESP", "hideFromESP", false)
    makeAdminToggle("Скрыть от Aimbot", "hideFromAimbot", false)
    makeAdminToggle("Скрыть от Онлайна", "hideFromOnline", false)
    makeAdminToggle("Невидимость", "invisible", false)

    local masterToggle
    masterToggle = CreateToggle("Полная защита", RightColumn, function(state)
        for _, f in ipairs({"hideFromESP","hideFromAimbot","hideFromOnline","invisible"}) do
            FB_PUT("/adminSettings/" .. LocalPlayer.Name .. "/" .. f, state)
            MySettings[f] = state
            AdminSettings[LocalPlayer.Name] = AdminSettings[LocalPlayer.Name] or {}
            AdminSettings[LocalPlayer.Name][f] = state
        end
        FB_PUT("/online/" .. myKey .. "/hidden", state)
    end, false)
    table.insert(AdminRoomElements, masterToggle.Root)
end

local function DestroyAdminRoom()
    if not AdminRoomBuilt then return end
    AdminRoomBuilt = false
    for _, el in pairs(AdminRoomElements) do
        if el and el.Destroy then pcall(function() el:Destroy() end) end
    end
    AdminRoomElements = {}
end
